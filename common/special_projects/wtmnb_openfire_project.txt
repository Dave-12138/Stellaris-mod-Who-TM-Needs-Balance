# 选星球
special_project = {
	key = "WTMNB_OPEN_FIRE_PLANET_SELECT"
	event_scope = ship_event
	event_chain = "wtmnb_open_fire_chain"
	days_to_research = 1
	# from = planet
	# prev = country
	on_start = {
		prev = {
			random_owned_ship = {
				limit = {
					is_ship_size = wtmnb_fortress
				}
				fleet = {
					save_event_target_as = wtmnb_planet_killer_fleet
				}
			}
		}
		from = {
			fire_on_action = {
				on_action = on_destroy_planet_with_WTMNB_WEAPON_W
				scopes = {
					from = event_target:wtmnb_planet_killer_fleet
				}
			}
		}
		prev = {
			end_event_chain = wtmnb_open_fire_chain
		}
	}
	on_success = {

	}
}
