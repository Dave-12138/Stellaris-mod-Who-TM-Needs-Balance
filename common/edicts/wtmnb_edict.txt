# 生成王座
wtmnb_edict_create_ship = {
	icon = GFX_ship_counter_32
	length = -1
	resources = {
		category = edicts
		upkeep = {
			alloys = 1
		}
	}
	effect = {
		# create_ship_design = {
		# 	design = "DESIGN_wtmnb_fortress"
		# }
		# add_ship_design = last_created_design
		if = {
			limit = {
				exists = capital_scope
			}
			capital_scope = {
				save_event_target_as = wtmnb_create_pos
			}
		}
		else_if = {
			limit = {
				count_system_within_border = {
					count > 0
				}
			}
			random_planet_within_border = {
				limit = {
					is_star = yes
				}
				save_event_target_as = wtmnb_create_pos
			}
		}
		else = {
			random_owned_fleet = {
				solar_system.star = {
					save_event_target_as = wtmnb_create_pos
				}
			}
		}
		event_target:wtmnb_create_pos = {
			create_fleet = {
				settings = {
					# can_upgrade = no
					spawn_debris = no
					# can_change_composition = no
					uses_naval_capacity = no
				}
				effect = {
					set_owner = root
					if = {
						limit = {
							root = {
								has_existing_ship_design = wtmnb_fortress
							}
						}
						create_ship = {
							random_existing_design = wtmnb_fortress
							name = wtmnb_fortress
							effect = {
								ship_event = {
									id = wtmnb_starbase.1000
									days = 1
								}
							}
						}
					}
					else = {
						create_ship = {
							name = wtmnb_fortress
							design = "DESIGN_wtmnb_fortress"
							effect = {
								ship_event = {
									id = wtmnb_starbase.1000
									days = 1
								}
							}
						}
					}
					set_location = prev
					set_fleet_stance = passive
					set_aggro_range_measure_from = self
					set_aggro_range = 350
				}
			}
		}
	}
	potential = {
		has_country_flag = wtmnb_order
		NOT = {
			any_owned_ship = {
				is_ship_size = wtmnb_fortress
			}
		}
		is_ai = no
	}
	allow = {
		is_ai = no
	}
}
