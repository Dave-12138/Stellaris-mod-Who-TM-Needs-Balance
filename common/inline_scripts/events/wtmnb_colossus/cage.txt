option = {
    name = wtmnb_colossus.1001.cage
    trigger = {
        from = {
            OR = {
                is_capital = yes
                AND = {
                    solar_system = {
                        is_capital_system = yes
                    }
                    is_star = yes
                }
            }
        }
    }
    custom_tooltip = wtmnb_colossus.1001.cage.tip
    hidden_effect = {
        from.solar_system = {
            every_system_planet_colony = {
                set_carrier_flag = in_capital_system
            }
            owner = {
                # 摧毁不在母星系的殖民地
                every_owned_colony = {
                    limit = {
                        not = { has_carrier_flag = in_capital_system }
                    }
                    every_owned_pop_group = {
                        kill_all_pop = yes
                    }
                    destroy_colony = yes
                    remove_carrier_flag = in_capital_system
                }
                # 恒星基地
                every_owned_starbase = {
                    limit = {
                        solar_system = {
                            NOT = {
                                is_same_value = prevprevprev
                            }
                        }
                    }
                    destroy_fleet = fleet
                }
                # 舰队遣返
                every_owned_fleet = {
                    limit = {
                        solar_system = {
                            NOT = {
                                is_same_value = prevprevprev
                            }
                        }
                        can_go_mia = yes
                    }
                    set_mia = mia_return_home
                }
            }
            # 封锁航道
            wtmnb_remove_hyperlane = yes
            # 加锁
            wtmnb_lock_bypass = {
                TIME = -1
            }
        }
    }
}
