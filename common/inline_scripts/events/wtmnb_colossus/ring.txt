#建造环世界
option = {
	name = wtmnb_colossus.1001.ring
	trigger = {
		from = {
			is_star = yes
			solar_system = {
				# 如果存在环则不能建，防止叠叠乐
				NOR = {
					any_system_megastructure = {
						OR = {
							is_megastructure_type = ring_world_1
							is_megastructure_type = ring_world_2_intermediate
							is_megastructure_type = ring_world_2
							is_megastructure_type = ring_world_3_intermediate
							is_megastructure_type = ring_world_ruined
							is_megastructure_type = ring_world_restored
						}
					}
					any_system_planet = {
						is_ringworld = yes
					}
				}
				OR = {
					# 不能有第二个恒星
					NOT = {
						any_system_planet = {
							is_star = yes
							NOT = {
								is_same_value = prevprev
							}
						}
					}
					# 除非 别的是炸出来的
					has_star_flag = single_blackhole
				}
			}
		}
		event_target:wtmnb_planet_killer_fleet = {
			any_owned_ship = {
				has_component = WTMNB_1ST_KEY
				has_component = WTMNB_4TH_KEY
				has_component = WTMNB_6TH_KEY
				has_component = WTMNB_9TH_KEY
			}
		}
	}
	from = {
		solar_system = {

			hidden_effect = {

				every_system_planet = {
					limit = {
						NOT = {
							is_same_value = prev.star
						}
					}
					remove_planet = yes
				}
				spawn_megastructure = {
					type = "ring_world_2_intermediate"
					planet = star
					owner = prevprev
				}
				every_system_megastructure = {
					limit = {
						is_megastructure_type = ring_world_2
					}
					upgrade_megastructure_to = ring_world_3_intermediate
				}
				spawn_megastructure = {
					type = "dyson_swarm_3"
					planet = star
					owner = prevprev
				}
				star = {
					if = {
						limit = {
							has_deposit_for = shipclass_mining_station
						}
						create_mining_station = {
							owner = root
						}
					}
					else = {
						create_research_station = {
							owner = root
						}
					}
				}
			}
		}
	}
	custom_tooltip = "wtmnb_colossus.1001.ring.desc"
}
