# 轨道轰炸-陆军
planet_event = {
    id = wtmnb.1000
    pre_triggers = {
        has_owner = yes
    }
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        has_orbital_bombardment_stance = wtmnb_invasion
        FROM = {
            has_country_flag = wtmnb_order
        }
    }
    immediate = {
        owner = {
            save_event_target_as = doomed_planet_owner
        }
        every_planet_army = {
            limit = {
                is_owned_by = prev.owner
            }
            remove_army = yes
        }
        if = {
            limit = { pop_amount > 0 }
            create_army = {
                owner = from
                # species = from
                type = "sentinel_army"
            }
        }
    }
}

# 轨道轰炸
planet_event = {
    id = wtmnb.1010
    pre_triggers = {
        has_owner = yes
    }
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        has_orbital_bombardment_stance = wtmnb_doom
        FROM = {
            has_country_flag = wtmnb_order
        }
        NOT = {
            is_planet_class = pc_ai
        }
    }
    immediate = {
        owner = {
            save_event_target_as = doomed_planet_owner
        }
        #remove_all_buildings = yes
        random_owned_pop_group = {
            kill_single_pop = yes
        }
        every_planet_army = {
            remove_army = yes
        }
        if = {
            limit = {
                is_colony = no
            }
            event_target:doomed_planet_owner = {
                add_static_war_exhaustion = {
                    attacker = from
                    location = root
                    value_for_planet_destruction = 1
                }
                #country_event = { id = wtmnb.1 }
            }
            from = {
                country_event = {
                    id = wtmnb.1
                }
            }
        }
        remove_modifier = "dangerous_wildlife"
    }
}
