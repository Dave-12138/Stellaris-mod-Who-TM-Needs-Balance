namespace = wtmnb_ships
# 所有王座造的船
ship_event = {
	id = wtmnb_ships.1000
	is_triggered_only = yes
	hide_window = yes
	trigger = {
		from = {
			is_ship_size = wtmnb_fortress
		}
	}
	immediate = {
		# 群星修仙传
		set_variable = {
			which = sxx_ship_state
			value = 10
		}
	}
}


#跃迁失踪
ship_event = {
	id = wtmnb_ships.2000
	is_triggered_only = yes
	hide_window = yes
	trigger = {
		solar_system = {
			has_star_flag = wtmnb_star_11th
		}
		owner = {
			NOT = {
				has_country_flag = wtmnb_order
			}
		}
	}
	immediate = {
		root.fleet = {
			set_mia = mia_return_home
			# set_mia_return_delay = 30
		}
	}
}
