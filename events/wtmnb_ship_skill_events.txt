namespace = wtmnb_ship_skill
# 切换船坞
fleet_event = {
	id = wtmnb_ship_skill.1000
	trigger = {
		from = {
			is_ship_size = wtmnb_fortress
		}
	}
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		from.starbase = {
			if = {
				limit = {
					has_starbase_flag = use_beastport
				}
				remove_starbase_flag = use_beastport
				set_starbase_flag = no_shipyard
				set_ship_construction_type = none
			}
			else_if = {
				limit = {
					has_starbase_flag = no_shipyard
				}
				remove_starbase_flag = no_shipyard
				set_ship_construction_type = starbase_shipyard
			}
			else = {
				set_starbase_flag = use_beastport
				set_ship_construction_type = starbase_beastport
			}
		}
	}
}

# 清除动作
event = {
	id = wtmnb_ship_skill.2000
	trigger = {
		from = {
			is_ship_size = wtmnb_fortress
		}
	}
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		from = {
			clear_orders = yes
			# clear_fleet_actions = this
		}
	}
}

# 升级按钮
fleet_event = {
	id = wtmnb_ship_skill.3000
	trigger = {
		from = {
			is_ship_size = wtmnb_fortress
		}
	}
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		from = {
			set_fleet_settings = {
				can_upgrade = yes
				spawn_debris = no
				uses_naval_capacity = no
			}
		}
	}
}
