namespace = wtmnb_ship_skill
# 切换船坞
fleet_event = {
	id = wtmnb_ship_skill.1000
	trigger = {
		from = {
			is_ship_size = wtmnb_fortress
		}
	}
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		from.starbase = {
			if = {
				limit = {
					has_starbase_flag = use_beastport
				}
				remove_starbase_flag = use_beastport
				set_starbase_flag = no_shipyard
				set_ship_construction_type = none
			}
			else_if = {
				limit = {
					has_starbase_flag = no_shipyard
				}
				remove_starbase_flag = no_shipyard
				set_ship_construction_type = starbase_shipyard
			}
			else = {
				set_starbase_flag = use_beastport
				set_ship_construction_type = starbase_beastport
			}
		}
	}
}

# 升级按钮
fleet_event = {
	id = wtmnb_ship_skill.3000
	trigger = {
		from = {
			is_ship_size = wtmnb_fortress
		}
	}
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		from = {
			set_fleet_settings = {
				can_upgrade = yes
				spawn_debris = no
				uses_naval_capacity = no
			}
		}
	}
}

# 工程船拆巨构
fleet_event = {
	id = wtmnb_ship_skill.10000
	trigger = {
		from = {
			is_ship_size = wtmnb_constructor
			solar_system? = {
				has_any_megastructure = yes
			}
		}
	}
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		from = {
			begin_event_chain = {
				event_chain = wtmnb_remove_megastructure_chain
				target = controller
			}
			controller = {
				set_variable = {
					which = wtmnb_rmmega_chain_count
					value = 0
				}
			}
			solar_system = {
				every_system_megastructure = {
					enable_special_project = {
						name = WTMNB_REMOVE_MEGASTRUCTURE
						location = this
						owner = from.controller
					}
					from.controller = {
						change_variable = {
							which = wtmnb_rmmega_chain_count
							value = 1
						}
					}
				}
			}
		}
	}
}
